**Goal:** Build a self-contained `tests/` directory with Playwright E2E tests covering 174 UX scenarios and three `node:assert` unit test files, runnable via `npm test` with no external dependencies beyond what's installed.

**Architecture:** Playwright starts the Express server via `webServer` config (port 3001, `DB_PATH=/tmp/queuec-test.db`). A global setup script seeds 20 fixture players into the test DB before any suite runs. E2E specs share a `helpers.js` module for common operations (create session, check in N players, start session). Unit tests are plain Node scripts run via a shell script.

**Tech Stack:** `@playwright/test` (already installed globally), `node:assert` (stdlib), Express + better-sqlite3 (existing server), Vite (existing client — NOT used in tests; server serves built `public/`).

- All test files live under `tests/` — no modifications to `client/` or `server/` source except one env-var addition to `server/db.js` (Task 1).

- [ ] **Step 1: Create `tests/helpers.js`**

  ```js
  // tests/helpers.js
  export const BASE = 'http://localhost:3001'

  const DEFAULT_PLAYERS = [
    'Inter C1', 'Inter C2', 'Inter C3', 'Inter C4',
    'Advanced E1', 'Advanced E2', 'Upper D1', 'Upper D2',
  ]

  export async function createSession(page, { name = 'Test Session', courts = 2 } = {}) {
    await page.goto('/')
    await page.getByRole('button', { name: '+ Session' }).click()
    await page.getByLabel('Name').fill(name)
    await page.getByLabel('Courts').fill(String(courts))
    await page.getByRole('button', { name: 'Create Session' }).click()
    // Modal closes reactively via SSE
    await page.waitForSelector('text=Not started', { timeout: 5000 })
  }

  export async function checkinPlayers(page, names = DEFAULT_PLAYERS) {
    for (const name of names) {
      await page.getByRole('button', { name, exact: false }).first().click()
      await page.waitForTimeout(200) // allow SSE to propagate
    }
  }

  export async function startSession(page) {
    await page.getByRole('button', { name: 'Start Session' }).click()
    await page.waitForSelector('[data-testid="courts-row"], text=Queue', { timeout: 5000 })
  }

  export async function fullSetup(page, { courts = 2, playerNames = DEFAULT_PLAYERS } = {}) {
    await createSession(page, { courts })
    await checkinPlayers(page, playerNames)
    await startSession(page)
  }

  export async function endMatch(page, courtLabel, { team1 = 21, team2 = 15 } = {}) {
    const courtCard = page.locator(`text=${courtLabel}`).locator('../..')
    await courtCard.getByRole('button', { name: 'End Match' }).click()
    await page.getByLabel('Team 1').fill(String(team1))
    await page.getByLabel('Team 2').fill(String(team2))
    await page.getByRole('button', { name: 'Confirm' }).click()
    await page.waitForTimeout(300)
  }

  export async function getQueueNames(page) {
    return page.locator('[data-entry-id]').evaluateAll(els =>
      els.map(el => el.querySelector('.text-base.font-medium')?.textContent?.trim()).filter(Boolean)
    )
  }
  ```

  ```js
  // tests/unit/timer.test.js
  import assert from 'node:assert/strict'

  function computeTimers(serverTime, queue, activeMatches) {
    const waitSeconds = new Map()
    for (const e of queue ?? []) {
      if (e.status === 'waiting') {
        waitSeconds.set(e.id, Math.max(0, serverTime - e.checked_in_at - e.total_seconds_played))
      }
    }
    const playSeconds = new Map()
    for (const m of activeMatches ?? []) {
      playSeconds.set(m.id, Math.max(0, serverTime - m.started_at))
    }
    return { waitSeconds, playSeconds }
  }

  const now = 1000

  // U3.1 — waiting player wait time
  const { waitSeconds: w1 } = computeTimers(now,
    [{ id: 1, status: 'waiting', checked_in_at: 400, total_seconds_played: 0 }], [])
  assert.equal(w1.get(1), 600, 'U3.1 wait = serverTime - checked_in_at - played')

  // U3.2 — playing player not in waitSeconds
  const { waitSeconds: w2 } = computeTimers(now,
    [{ id: 2, status: 'playing', checked_in_at: 300, total_seconds_played: 200 }], [])
  assert.equal(w2.has(2), false, 'U3.2 playing player absent from waitSeconds')

  // U3.3 — active match play time
  const { playSeconds: p3 } = computeTimers(now, [], [{ id: 10, started_at: 900 }])
  assert.equal(p3.get(10), 100, 'U3.3 play = serverTime - started_at')

  // U3.4 — negative clamped to 0
  const { waitSeconds: w4 } = computeTimers(100,
    [{ id: 3, status: 'waiting', checked_in_at: 50, total_seconds_played: 200 }], [])
  assert.equal(w4.get(3), 0, 'U3.4 negative result clamped to 0')

  // U3.5 — null/undefined handled gracefully
  const { waitSeconds: w5, playSeconds: p5 } = computeTimers(now, null, undefined)
  assert.equal(w5.size, 0, 'U3.5a null queue → empty map')
  assert.equal(p5.size, 0, 'U3.5b undefined matches → empty map')

  console.log('timer.test.js: all assertions passed ✓')
  ```
